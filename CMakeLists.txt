cmake_minimum_required(VERSION 3.16)
project(ignition_zero LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_library(ignition_zero_src STATIC src/ignition_zero.cpp)
target_include_directories(ignition_zero_src PUBLIC src)
if(MSVC)
    target_compile_options(ignition_zero_src PRIVATE /W4)
else()
    target_compile_options(ignition_zero_src PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(test_ignition tests/test_ignition.cpp)
target_link_libraries(test_ignition PRIVATE ignition_zero_src)
if(MSVC)
    target_compile_options(test_ignition PRIVATE /W4)
else()
    target_compile_options(test_ignition PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME ignition_checks COMMAND test_ignition)

add_executable(ignition_demo src/ignition_demo.cpp)
target_link_libraries(ignition_demo PRIVATE ignition_zero_src)
if(MSVC)
    target_compile_options(ignition_demo PRIVATE /W4)
else()
    target_compile_options(ignition_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()
